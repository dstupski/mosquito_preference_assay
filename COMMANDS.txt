===============================================================================
 mosquito_preference_assay -- copy-paste command sheet
===============================================================================
 Every block below is meant to be pasted straight into a terminal.
 Lines starting with # are comments and are safe to paste along with the rest.

 EVERY NEW TERMINAL needs this first:

     source ~/ros2_ws/install/setup.bash

 (on the rig PC the workspace is ~/mosquito_ros2_ws -- adjust the path)
===============================================================================


-------------------------------------------------------------------------------
 0. FIRST TIME ON THIS MACHINE  (or after any git pull)
-------------------------------------------------------------------------------

cd ~/ros2_ws
git -C src/mosquito_preference_assay pull
colcon build --packages-select mosquito_preference_assay --symlink-install
source install/setup.bash

# --symlink-install is NOT optional. Without it colcon COPIES config/ and
# experiments/ into install/, your *.local.yaml files are never found, and
# edits you make in src/ silently do nothing. Verify -- you want to see
# arrows (->) pointing into src/:

ls -l install/mosquito_preference_assay/share/mosquito_preference_assay/config/

# After RENAMING or DELETING a file, a plain rebuild is not enough (symlinks
# are never removed, so stale names linger as dead links). Clean rebuild:

rm -rf build/mosquito_preference_assay install/mosquito_preference_assay
colcon build --packages-select mosquito_preference_assay --symlink-install
source install/setup.bash


-------------------------------------------------------------------------------
 1. ONE-TIME RIG SETUP  (repeat only if the projector or a camera moves)
-------------------------------------------------------------------------------

# --- 1a. make your own config, which git will never overwrite ---------------
cd ~/ros2_ws/src/mosquito_preference_assay
cp config/assay_params.yaml    config/assay_params.local.yaml
cp config/detector_params.yaml config/detector_params.local.yaml

# In config/detector_params.local.yaml, the two that matter most:
#     image_topic: "/cam_sync/cam0/image_raw"   <- which camera detects
#     polarity: "darker"                        <- only DARK objects fire,
#                                                  so reflections/LEDs cannot
#
# Then edit config/assay_params.local.yaml and set AT LEAST:
#     monitor: "2"            <- quoted; a bare 2 is a parameter-type error
#     fullscreen: true        <- monitor is IGNORED unless this is true
#     stimuli_when_armed: true  <- if you want no stimulus onset (see 3b)


# --- 1b. which display is the projector? ------------------------------------
python3 ~/ros2_ws/src/mosquito_preference_assay/tools/list_displays.py --identify

# Puts a big number on each physical screen in turn. Use THAT number --
# Ubuntu's Displays panel numbers them differently.


# --- 1c. place the stimulus circles on the projector ------------------------
ros2 launch mosquito_preference_assay display_check.launch.py

#   drag the midpoint .... move the pair
#   drag a circle ........ change their separation
#   [  ]  ................ resize both
#   s  ................... SAVE   <- this is the whole handoff
#   q  ................... quit
#
# `s` writes config/display_geometry.local.yaml, which every launch picks up
# automatically. Nothing to copy anywhere.


# --- 1d. draw the trigger zone on the live camera ---------------------------
# The camera must already be publishing (see section 2).

ros2 launch mosquito_preference_assay trigger_roi.launch.py               # configured camera
ros2 launch mosquito_preference_assay trigger_roi.launch.py camera:=cam1  # this session only

# To change the camera PERMANENTLY, edit config/detector_params.local.yaml:
#     image_topic: "/cam_sync/cam1/image_raw"
# That same file is what the real detector reads, so the tool and the
# experiment can never end up pointed at different cameras.

#   drag inside the box .. move the zone
#   drag a corner ........ resize
#   drag empty image ..... draw a new zone
#   [  ]  ................ shrink / grow
#   d  ................... detection overlay on/off (green = would fire)
#   b  ................... re-capture the background frame
#   s  ................... SAVE
#   q  ................... quit
#
# `s` writes config/trigger_roi.local.yaml, including WHICH CAMERA the zone
# was drawn on -- so doing this on cam0 is what points detection at cam0.


-------------------------------------------------------------------------------
 2. START THE CAMERAS   (your own stereo package -- not part of this repo)
-------------------------------------------------------------------------------

# >>> REPLACE THIS LINE with your camera package's launch command <<<
ros2 launch <your_camera_package> <your_camera_launch>.py

# Confirm both feeds are live before running anything else. A launch against
# silent cameras comes up armed and simply waits forever:

ros2 topic hz /cam_sync/cam0/image_raw
ros2 topic hz /cam_sync/cam1/image_raw


-------------------------------------------------------------------------------
 3. RUN THE SIPPELL RETEST  <<< THE MAIN EVENT >>>
-------------------------------------------------------------------------------

 TERMINAL 1 -- cameras (section 2), left running.
 TERMINAL 2 -- one animal, one trial, one run folder:

source ~/ros2_ws/install/setup.bash
ros2 launch mosquito_preference_assay arena_experiment.launch.py \
    experiment_file:=sippell_retest_experiment \
    save_dir:=/data/mosquito/$(date +%F) \
    stimuli_when_armed:=true

# That is the whole run. It picks up your projector alignment, your trigger
# zone, your monitor and your detection camera on its own -- everything
# calibrated in section 1 is applied without being named here.
#
# It ends by itself: trial runs 15 s, the sketch exits, both bags close.
# One launch = one animal = one run folder. Re-run it for the next animal.


# --- 3a. VERIFY BY EYE which stimulus is on which side ----------------------
# In a spare terminal, while the run is up. With stimuli_when_armed: true the
# sides are known while still ARMED, so you can stand at the arena and check
# the projection before an animal is introduced.

python3 ~/ros2_ws/src/mosquito_preference_assay/tools/watch_stimuli.py

#   armed     LEFT=blank   RIGHT=telescope_inward
#   running   LEFT=blank   RIGHT=telescope_inward   trial_seed=2873312514

# The raw topics, if you would rather read the JSON yourself:
ros2 topic echo /stimulus_publisher/stimulus_state     # continuous, 10 Hz
ros2 topic echo /stimulus_publisher/trial_start        # once, at trial start
ros2 topic echo /stimulus_publisher/stimulus_position  # ~60 Hz, x/y of each
                                                       # stimulus as drawn

# WHERE THE JITTERING TARGET ACTUALLY WAS, per frame -- this is what you
# relate a flight track to. Recorded in the assay bag, from the trigger on.


# --- 3b. what you should see -----------------------------------------------
#   run folder      : /data/mosquito/2026-09-28/trial_20260928_101500
#   trigger zone    : .../config/trigger_roi.local.yaml
#   detection camera: /cam_sync/cam0/image_raw
#   [assay] display 2: 1920x1080 ...              <- the projector
#   [assay] ARMED -- stimuli are PLAYING; the trigger starts the recording window
#   mosquito_detector: holding fire (assay phase=None)
#   mosquito_detector: assay phase: armed
#   mosquito_detector: mosquito_detected at (...)
#   [assay] trial 0 START (stimuli already playing 7.9s)


# --- 3c. NEXT ANIMAL, without restarting ------------------------------------
# When a trial finishes, swap the animal and run this in a SECOND terminal:

ros2 run mosquito_preference_assay rearm

# A new pairing is drawn and starts playing; the detector re-arms itself.
# Each trial writes its OWN folder, named at the trigger:
#     <save_dir>/trial_20261007_103440/{assay,video}
#     <save_dir>/trial_20261007_103527/{assay,video}
#
# Refused while a trial is still running -- wait for it to finish.
# Ctrl-C the launch when you are done for the session.


# --- 3d. how a run begins and ends ------------------------------------------
# Recording starts AT THE TRIGGER -- nothing is written while waiting for an
# animal. BOTH bags close when the trial ends, automatically.
#
# The display then STAYS UP showing the background (hold_after_trial, on by
# default), so the arena's light does not change between animals. Nothing is
# recorded during that time. Ctrl-C when you are ready for the next animal.
#
# For a run that terminates by itself instead, add:
#     hold_after_trial:=false


# --- 3e. without the always-on stimuli (circles appear at detection) --------
ros2 launch mosquito_preference_assay arena_experiment.launch.py \
    experiment_file:=sippell_retest_experiment \
    save_dir:=/data/mosquito/$(date +%F)


# --- 3f. useful variations --------------------------------------------------

# name the run folder after the animal
ros2 launch mosquito_preference_assay arena_experiment.launch.py \
    experiment_file:=sippell_retest_experiment \
    save_dir:=/data/mosquito/$(date +%F) run_name:=mosq07 \
    stimuli_when_armed:=true

# detect on the OTHER camera for one run
    detection_camera:=cam1

# no video (assay topics only -- small bags, quick tests)
    record_video:=false

# no recording at all (just watch it happen)
    record:=false

# reproduce an earlier run exactly (master_seed is printed in every run's log
# and stored in the bag's experiment_info)
    master_seed:=123456789

# use a specific older calibration instead of the current one
    display_config:=config/20260923_display_config.yaml
    trigger_config:=config/20260924_trigger_roi.yaml

-------------------------------------------------------------------------------
 7. WHEN SOMETHING IS WRONG
-------------------------------------------------------------------------------

# --- display opens on the wrong screen --------------------------------------
# monitor is IGNORED unless fullscreen is true. Both must be set:
grep -n "monitor\|fullscreen" ~/ros2_ws/src/mosquito_preference_assay/config/assay_params.local.yaml
# The sketch prints where it actually landed -- trust this, not the config:
#     [assay] display 2: 1920x1080 at +1920+0 (:0.1)


# --- edits to a config file seem to do nothing ------------------------------
# Almost always a copy build. Regular files instead of symlinks = broken:
ls -l ~/ros2_ws/install/mosquito_preference_assay/share/mosquito_preference_assay/config/
# Fix: the clean rebuild in section 0.


# --- which files did the launch actually load? ------------------------------
# Every launch prints this at startup. Read it before assuming anything:
#     display calibration: .../config/display_geometry.local.yaml
#     trigger zone       : .../config/trigger_roi.local.yaml
#     detection camera   : /cam_sync/cam0/image_raw


# --- "invalid YAML: mapping values are not allowed here" --------------------
cd ~/ros2_ws/src/mosquito_preference_assay
python3 - <<'PY'
import glob, yaml
for path in sorted(glob.glob("config/*.yaml") + glob.glob("experiments/*.yaml")):
    try:
        yaml.safe_load(open(path)); print(f"  ok   {path}")
    except yaml.YAMLError as e:
        mark = getattr(e, "problem_mark", None)
        print(f"  FAIL {path}\n       {getattr(e,'problem','')}")
        if mark:
            lines = open(path).read().splitlines()
            for n in range(max(0, mark.line-2), min(len(lines), mark.line+2)):
                print(f"       {'>>' if n==mark.line else '  '} {n+1:4d} | {lines[n]}")
PY


# --- it never triggers ------------------------------------------------------
ros2 topic hz /cam_sync/cam0/image_raw      # camera actually publishing?
ros2 topic echo /arena/mosquito_present     # detector firing at all?
# "holding fire (assay phase=None)" is CORRECT during the first few seconds --
# the detector waits for the display to finish booting so the animal's first
# approach is not wasted.


# --- it triggers on nothing -------------------------------------------------
# The trigger zone is too big, or includes equipment. Redraw it (section 1d)
# and watch the overlay: green = would fire, grey = seen and ignored.


# --- stray processes from an earlier test still publishing ------------------
ps -eo pid,etimes,args --no-headers | grep -E "video_publisher|bag record" | grep -v grep
# then: kill -INT <pid>
